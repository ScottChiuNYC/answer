cmake_minimum_required(VERSION 3.20)
project(answer VERSION 0.1.0 LANGUAGES CXX)
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
add_library(answer STATIC src/answer.cpp)
add_library(answer::answer ALIAS answer)
target_compile_features(answer PUBLIC cxx_std_17)
target_include_directories(answer PUBLIC "$<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include>" "$<INSTALL_INTERFACE:${CMAKE_INSTALL_INCLUDEDIR}>")
set_target_properties(answer PROPERTIES EXPORT_NAME answer)
install(TARGETS answer EXPORT answerTargets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(DIRECTORY include/ DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT answerTargets NAMESPACE answer:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/answer)
configure_package_config_file(cmake/answerConfig.cmake.in "${CMAKE_CURRENT_BINARY_DIR}/answerConfig.cmake" INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/answer)
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/answerConfigVersion.cmake" VERSION ${PROJECT_VERSION} COMPATIBILITY SameMajorVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/answerConfig.cmake" "${CMAKE_CURRENT_BINARY_DIR}/answerConfigVersion.cmake" DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/answer)
enable_testing()
add_executable(answer_test tests/answer_test.cpp)
target_link_libraries(answer_test PRIVATE answer::answer)
add_test(NAME answer_returns_42 COMMAND answer_test)
